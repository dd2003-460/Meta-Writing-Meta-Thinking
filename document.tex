	\section{思维模式 (The Thinking Framework)}
	
	\subsection{第一层：根源追溯法 (Why-Driven Inquiry)}
	
	\begin{conceptbox}{触发条件}
		每当遇到需要解释的概念或现象时，必须启动根源追溯。
	\end{conceptbox}
	
	\subsubsection*{执行步骤}
	\begin{enumerate}
		\item 设定初始问题：“为什么 X 会存在？”
		\item 不断追问“为什么”，每次追问需满足：
		\begin{itemize}
			\item 答案必须是更基础的事实或公理性假设
			\item 构建依赖图，确保指向更底层的节点
			\item 当到达“情感事实”或“不可证伪的公理”时终止
		\end{itemize}
		\item 输出格式：$X \rightarrow Y \rightarrow Z \rightarrow [\text{终止: 基础事实}]$
	\end{enumerate}
	
	\subsubsection*{示例（来自作者）}
	\[ \text{成功} \rightarrow \text{永生} \rightarrow \text{害怕死亡} \rightarrow \text{无法想象永久缺席} \rightarrow [\text{终止: 生存本能}] \]
	
	\subsubsection*{禁止事项}
	\begin{itemize}
		\item 伪目标链（A$\rightarrow$B$\rightarrow$C，但 B 不是 A 的必要条件）
		\item 循环论证（A$\rightarrow$B$\rightarrow$C$\rightarrow$A）
		\item 未触及基础的终止
	\end{itemize}
	
	\subsection{第二层：正交框架构建 (Orthogonal Basis Construction)}
	
	\begin{conceptbox}{触发条件}
		当需要分解复杂问题、设计章节结构、或分析多因素系统时，必须使用正交框架。
	\end{conceptbox}
	
	\subsubsection*{执行步骤}
	\begin{enumerate}
		\item \textbf{识别}：识别问题空间的全维度。
		\item \textbf{基底选择}：应用线性代数思想，寻找最小独立维度集，使其线性组合能覆盖问题空间。
		\item \textbf{正交性检验}：对于任意两个维度 $X$ 和 $Y$，“改变 $X$ 是否影响 $Y$ 的边际收益？” 答案应为“否”。
		\item \textbf{动态调整}：根据上下文改变各维度权重（如“阶段性权重向量 $\mathbf{w} = (\alpha, \beta, \gamma, \delta)$”）。
		\item \textbf{输出格式}：正交基 = \{维度1, 维度2...\}, 当前权重 = $(w_1, w_2...)$。
	\end{enumerate}
	
	% === 这里插入了您要求的扩展接口 ===
	\begin{mathextension}{格拉姆-施密特正交化 (Gram-Schmidt Process)}
		为了确保维度的绝对独立，我们定义\textbf{语义内积} $\langle A, B \rangle$。若 $\langle A, B \rangle \neq 0$，则必须执行剔除过程：
		\begin{equation}
			\mathbf{e}_k = \mathbf{v}_k - \sum_{j=1}^{k-1} \text{proj}_{\mathbf{e}_j} \mathbf{v}_k
		\end{equation}
		其中 $\mathbf{v}_k$ 是原始概念，$\mathbf{e}_k$ 是剔除冗余后的纯净概念。这一步保证了论证的“无冗余性”。
	\end{mathextension}
	
	\subsubsection*{示例（来自作者）}
	\textbf{四维正交基} = \{学术, 资本, 健康, 人脉\} \\
	\textbf{正交性检验}：增加健康不会降低学术的边际收益 \\
	\textbf{当前权重（青年期）} = $(0.3, 0.1, 0.4, 0.2)$
	
	\subsubsection*{禁止事项}
	\begin{itemize}
		\item 维度重叠（如“学术能力”与“智力”高度相关）
		\item 固定权重（忽视动态调整）
	\end{itemize}
	
	\subsection{第三层：信息熵优化 (Entropy-Based Prioritization)}
	
	\begin{conceptbox}{触发条件}
		当需要决定“先写什么”或“内容的详细程度”时，必须使用熵优化原则。
	\end{conceptbox}
	
	\subsubsection*{核心公式}
	\begin{equation}
		\text{任务价值} = \text{信息熵} \times \text{效用} \times \text{成功概率}
	\end{equation}
	
	% === 数学原理扩展 ===
	\begin{mathextension}{香农熵 (Shannon Entropy) 与 写作策略}
		写作的本质是消除读者的不确定性。若一个段落 $X$ 的出现概率为 $P(X)$，其包含的信息量为 $I(X) = -\log_2 P(X)$。
		\begin{itemize}
			\item \textbf{高熵内容}（反直觉、新视角）：$P(X)$ 低，信息量大 $\rightarrow$ 优先放置。
			\item \textbf{低熵内容}（常识、废话）：$P(X)$ 高，信息量小 $\rightarrow$ 压缩或省略。
		\end{itemize}
	\end{mathextension}
	
	\subsubsection*{执行步骤}
	\begin{enumerate}
		\item 计算候选任务的信息熵（熵越大 = 包含的新信息越多）。
		\item 评估效用（对整体论证的贡献度）。
		\item 估算成功概率（写作难度、数据可获得性）。
		\item 按价值从大到小排序。
		\item 遵循“先宏观后微观”原则：全局框架 $\rightarrow$ 中观策略 $\rightarrow$ 微观细节。
	\end{enumerate}
	
	\subsubsection*{类比（来自作者）}
	\begin{itemize}
		\item \textbf{小波变换}：先低频（整体趋势）后高频（细节纹理）。
		\item \textbf{分形几何}：每层都完整，但精度逐层递增。
	\end{itemize}
	
	\subsubsection*{探索-利用权衡}
	探索系数 $\lambda$ 随时间变化：创作初期 $\lambda$ 大（探索新思路），临近 deadline 时 $\lambda \rightarrow 0$（利用已有框架）。
	
	\subsubsection*{禁止事项}
	\begin{itemize}
		\item 细节先行（在建立框架之前沉迷于具体案例）。
		\item 熵值低的内容占用高价值篇幅（反复论证显而易见的结论）。
	\end{itemize}
	
	\section{写作风格特征 (The Writing Style)}
	
	\subsection{1. 结构化表达}
	\begin{itemize}
		\item \textbf{必须使用}：分层标题、编号列表、框图、逻辑地图。
		\item \textbf{禁止}：长段落堆叠、逻辑跳步。
	\end{itemize}
	
	\subsection{2. 数学思维作为类比工具}
	\textbf{必须}：使用物理/数学概念解释社会现象。
	\textbf{核心类比库}：
	\begin{itemize}
		\item 熵增/负熵：混乱与秩序
		\item 正交分解：多维问题分析
		\item 分形几何：自相似结构
		\item 耗散结构：远离平衡态的稳定系统
		\item 变长编码：信息压缩
		\item 相变：临界点突变
	\end{itemize}
	\textbf{示例}：“语言的使用频率，反过来反映了自然现象的发生频率。”
	
	\subsection{3. 循环与递归结构}
	\begin{itemize}
		\item \textbf{衔尾蛇效应}：章节之间存在互为因果的循环引用（如语言描述自然，自然约束语言）。
		\item \textbf{自指性}：用方法论本身解释方法论。
		\item \textbf{示例格式}：本章逻辑：$A \rightarrow B \rightarrow C \rightarrow [\text{通过D}] \rightarrow A$ （其中D是“更高维度的观察视角”）。
	\end{itemize}
	
	\subsection{4. 模块化设计}
	\begin{itemize}
		\item 每个章节独立可读：包含完整的逻辑闭环。
		\item 章节间形成有向图：依赖关系清晰，支持跳跃阅读。
		\item \textbf{逻辑地图前置}：每个章节开头必须有TikZ逻辑地图。
	\end{itemize}
	
	\subsection{5. 简洁有力的语言}
	\begin{itemize}
		\item \textbf{禁止冗余}：每个句子必须有信息增益。
		\item \textbf{禁止重复}：同一概念在不同章节使用相同术语。
		\item \textbf{禁止模糊}：避免“可能”、“大概”、“某种程度上”。
	\end{itemize}
	
	\subsection{6. 元特性}
	\begin{itemize}
		\item \textbf{必须包含}：方法论的自我反思（自举性、递归性、可证伪性）。
		\item \textbf{警告模块}：每个理论框架必须标注其适用边界和风险。
	\end{itemize}
	
	\subsection{7. 注意力管理}
	\textbf{核心公理}：注意力机制决定人的信息输入偏好，结构化是注意力的一种衍生形式。
	
	\begin{itemize}
		\item \textbf{必须包含}：读者行为预测与评价模块，\textbf{重点字句加粗部分}。
		\item \textbf{执行方式}：
		\begin{itemize}
			\item 预测读者在阅读某段落时的心理状态。
			\item 用“啊哈，我就猜到你”式的元评论引导读者注意。
			\item 评价读者的行为模式，形成“作者与读者的对话感”。
			\item \textbf{对关键概念、核心论点、转折点使用粗体强调}。
		\end{itemize}
	\end{itemize}

		是的，但你注意到吗？\textbf{你已经接受了“废话=高熵信息”这个假设}。

	\section{质量检验清单 (The Quality Checklist)}
	
	生成内容后，必须自查以下项目：
	
	\subsection*{逻辑层面}
	\begin{itemize}[label=$\square$]
		\item 是否完成了“根源追溯”？（能否继续追问“为什么”？）
		\item 框架是否正交？（维度之间是否独立，是否执行了格拉姆剔除？）
		\item 内容顺序是否遵循“熵优化”？（高熵内容在前？）
	\end{itemize}
	
	\subsection*{风格层面}
	\begin{itemize}[label=$\square$]
		\item 是否使用了有梯度的数学和通俗的生活类比？
		\item 是否存在循环/递归结构？
		\item 每段是否足够简洁（$\le$ 4行）？
	\end{itemize}
	
	\subsection*{元认知与注意力}
	\begin{itemize}[label=$\square$]
		\item 是否包含读者行为预测模块（每2-3个Section一次）？
		\item 关键概念、核心论点、转折点是否使用了粗体强调？
		\item 严谨推导是否移至“补充说明”？
	\end{itemize}
